# A song dropped on the window - the same path the drag & drop takes.
#
# The song is plugin/Tests/assets/minimal.vgm (generated, 431 bytes), which build.ps1 hands over as
# VGM_TEST_SONG; set that variable to any .vgm/.vgz/.s98/.dro/.gym file to run against something
# real. Without it the load is skipped and said so, rather than failed.
#
# Run:  set VGM_TEST_SONG=C:\music\song.vgz
#       build\plugin\vgm_app_test plugin\Tests\scripts\load-and-play.txt

load $VGM_TEST_SONG

# it plays, and the page that owns the cards is the one on screen
assert editor.status text contains "Playing"
assert view.play visible == true

# the chip card appeared, with one checkbox per chip (the song decides how many)
assert chip.0 visible == true

# A chip checkbox is a real toggle, so its state is worth asserting: ticked means the chip is
# audible. (The transport and mixer header buttons are NOT click-toggles - their panels light them
# in refresh(), which this runner does not drive, so `checked` reads "n/a" for them on purpose.)
assert chip.0 checked == true
click chip.0
assert chip.0 checked == false
click chip.0
assert chip.0 checked == true

# the transport answers the keys the transport card documents
press enter
press enter

# ... and the keys really reach the engine: a tick asks the window for one frame of refresh, and
# the status line is what the transport says afterwards (before `tick` it would still read the old
# text, which is why the command exists)
press space
tick
assert editor.status text contains "Paused"

# The tray menu acts on the engine, not on the panel's buttons. The panel must still show Pause,
# not fall through to Stop, because the engine is the one owner of the pause state.
press space
tick
assert btn.play checked == true
engine pause
tick
assert editor.status text contains "Paused"
assert btn.pause checked == true
assert btn.stop checked == false

# The menu does what the window does: this is the File > Close Song item, invoked by its id, with
# no menu ever opening. The song goes, the status line says so, and the page stays.
menu invoke file.close
tick
assert editor.status text contains "No file"
assert view.play visible == true

# the window still fits its contents after all of that
dump view.play
