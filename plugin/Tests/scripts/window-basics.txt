# The window as it starts, and the two switches every page has.
#
# Run:  build\plugin\vgm_app_test plugin\Tests\scripts\window-basics.txt
#
# No song and no sound device needed, so this runs anywhere - including CI.

# the chrome is always there
assert editor.menubar visible == true
assert editor.open    visible == true
assert editor.status  visible == true
assert editor.status  text contains "No file"

# the playback page is the one on screen
assert view.play       visible == true
assert view.visualiser visible == false

# the transport buttons are reachable by ID (IconButton names itself: "Play" -> btn.play)
assert btn.play  visible == true
assert btn.pause visible == true
assert btn.stop  visible == true
assert btn.loop  visible == true

# Ctrl+2 shows the visualisation page and Ctrl+1 brings the player back - the accelerators the View
# menu advertises, driven through the same keyPressed() the window itself uses
press ctrl+2
assert view.visualiser visible == true
assert view.play       visible == false
press ctrl+1
assert view.play       visible == true
assert view.visualiser visible == false

# space reaches the transport (the shortcut the transport card documents)
press space
