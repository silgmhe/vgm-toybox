# hooks audio
#
# The settings sidebar's pages, including the Audio page that only an application has.
# Without the line above this is the plugin's three-page panel, and the two Audio assertions fail.
#
# Run:  build\plugin\vgm_app_test plugin\Tests\scripts\settings-pages.txt
#
# The first line is read before the editor is constructed, because the panel decides whether it
# has an Audio page at that moment (see scriptAsksForAudioHooks in the harness). The hooks it
# installs are fakes: the device itself is not what this script is about.

# the panel opens on the first page, and the chrome is there
menu invoke options.settings
assert settings.close             visible == true
assert settings.page.appearance   visible == true
assert settings.appearance.font.value text contains "System UI"
# The face, not just the label: a default Font resolves through the default LookAndFeel, so this
# fails if the choice was applied only to the editor's own ToyboxLookAndFeel.
assert settings.appearance.font.value typeface == @systemUI
assert settings.appearance.font.choose enabled == true

# The Audio page was built and is not the page on screen. Existence is the point: the row and the
# page only exist where something filled in the hooks table before the panel was built.
assert settings.page.audio enabled == true
assert settings.page.audio visible == false

# The page reads the same table: the device line is the harness's fake, not a real device
assert settings.audio.device text contains "Fake device"

# The General page has the startup restore switch - the standalone's "last time" - and it is on by
# default. A plugin's panel does not build it, because a host project is what restores a plugin.
assert settings.general.restoreLastSong     enabled == true
assert settings.general.restoreLastSong     checked == true
assert settings.general.restoreLastSongHint text contains "Close Song"

# The tray switch sits under it, and is off by default: a close must keep quitting until the user
# asks the window to stay alive. The hint says what the close button will then do.
assert settings.general.closeToTray         enabled == true
assert settings.general.closeToTray         checked == false
assert settings.general.closeToTrayHint     text contains "tray"
toggle settings.general.closeToTray
assert settings.general.closeToTray         checked == true

# Switching through the real tree, not by setting a test-only page index: row 2 is Language in
# both the plugin and the application (rows are Appearance, General, Language, MIDI, then Audio).
select settings.tree 2
assert settings.page.language     visible == true
assert settings.page.appearance   visible == false

# Row 3 is the MIDI page. The bend range starts at the default 12 semitones each way, and bends
# are on: the switch is what turns the chips' fine pitch into 0xE0 at all.
select settings.tree 3
assert settings.page.midi         visible == true
assert settings.page.language     visible == false
assert settings.midi.bend         checked == true
assert settings.midi.range        value   ~= 12
assert settings.midi.range        enabled == true

# The engine holds both values, not just the page: the switch writes the shared settings file,
# the store broadcasts synchronously and the processor's change listener carries it to the audio
# side. Asserting on the page alone would not notice a listener that was never added.
assert engine bend                == true
assert engine bendRange           == 12
set settings.midi.range 7
assert settings.midi.range        value   ~= 7
assert engine bendRange           == 7
set settings.midi.range 12
assert engine bendRange           == 12

# Turning bends off greys the range but keeps the number: the row does not disappear, and the
# value is still what it was when the switch goes back on.
toggle settings.midi.bend
assert settings.midi.bend         checked == false
assert settings.midi.range        enabled == false
assert settings.midi.range        value   ~= 12
assert engine bend                == false
toggle settings.midi.bend
assert settings.midi.bend         checked == true
assert settings.midi.range        enabled == true
assert engine bend                == true
