# wrapper vst3
# The DAW-sync controls only exist in a hosted build (docs/desgin/daw-sync-cue-design.md, §7.4).
# They are one component (ui/SyncGroup), so the frame itself is in the tree as well.
assert group.sync visible == true
assert btn.sync visible == true
assert btn.cue  visible == true

# No host clock yet: there is nothing to take a cue from. The knob is live anyway - the cue always
# exists (0 is the DAW's start), so there is no "waiting for a cue" state to show.
assert btn.cue enabled == false
assert btn.cue checked == false
assert knob.cue-offset enabled == true
assert knob.cue-offset text == "0.0 ms"
assert btn.play enabled == true

# The host reports a play head - what a DAW wrapper does. A cue can be taken now.
host 12.5 playing
tick
assert btn.cue enabled == true
assert knob.cue-offset enabled == true
assert knob.cue-offset text == "0.0 ms"

# Taking the cue stores it and unlocks the knob, whose number is the offset itself. The button is
# never lit - a lit face means "this switch is on", and Cue is an action; the knob is the display.
click btn.cue
sleep 100
tick
assert btn.cue checked == false
assert knob.cue-offset enabled == true
assert knob.cue-offset text == "0.0 ms"

# Cue is not a toggle: clicking it again re-anchors the cue where the host is now. (The right-click
# menu on the box can move it back to 0 s, which a script cannot reach - a context menu is modal.)
# The offset survives the re-take: it is a trim on the cue, not part of it.
click btn.cue
sleep 100
tick
assert btn.cue checked == false
assert knob.cue-offset enabled == true

# The number beside the knob is the exact-value field - the same edit box the volume's "100 %" is,
# so this is also what typing into it does.
set knob.cue-offset 2.5
assert knob.cue-offset text == "+2.5 ms"

# Sync on: enabled + a usable host means the host drives the position, so the local transport is
# disabled - a button the next audio block would overwrite reads as broken. Only Sync lights up:
# Cue stays dark, it was the action that set the anchor, not a switch.
click btn.sync
sleep 100
tick
assert btn.sync checked == true
assert btn.cue checked == false
assert btn.play enabled == false
assert btn.stop enabled == false
assert btn.playback-start enabled == false
# the loop count, the fade and the volume are part of the mapping, so they stay live
assert btn.loop enabled == true

# And the space bar is the host's: "unhandled" is the window letting the key out instead of
# swallowing it, which is what hands it to the DAW's own transport shortcut. Consuming it here
# could only make it do nothing - the local transport is disabled above.
press space unhandled

# Sync off: the local transport comes back, and the anchor is still there (it always is) - it is
# just not being followed, so the light goes out and only the knob still shows the trim.
click btn.sync
sleep 100
tick
assert btn.sync checked == false
assert btn.play enabled == true
assert btn.stop enabled == true
assert btn.cue checked == false
assert knob.cue-offset enabled == true
assert knob.cue-offset text == "+2.5 ms"

# Nothing had to be re-bound: with the host no longer driving, the space bar is the local
# transport again - the same check that let it out says it is ours now.
press space
