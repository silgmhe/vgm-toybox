# wrapper vst3
# What the UI scripts cannot show: with the host driving, the *position* is the mapping
# (host time - cue), and that number lives in the engine, not in a control the window moves.
# `assert engine position` reads the transport itself, in seconds, so this is the end-to-end
# assertion of the feature - the one that would have caught a seek and a transport start being
# applied in the wrong order (docs/review/2026-10-17-..., H-1).
#
# Needs a song: build.ps1 -Test hands over plugin/Tests/assets/minimal.vgm (6 s, enough for the
# positions below). Without VGM_TEST_SONG the load is skipped, and so is everything below it.
#   set VGM_TEST_SONG=C:\music\song.vgz
#   build\plugin\vgm_app_test plugin\Tests\scripts\daw-sync-follow.txt

load $VGM_TEST_SONG

# The host is playing at 2 s, which is enough for CUE to be enabled (it needs a usable clock).
host 2 playing
tick
assert btn.cue enabled == true

# CUE = 2 s, and sync on: from here the song's position is "host time - 2 s".
click btn.cue
sleep 100
click btn.sync
sleep 100

# A host parked *at* the cue: the song sits at its beginning, silent.
host 2 stopped
tick
assert engine position ~= 0
assert btn.play checked == false

# One second later the song is one second in. This is the transition that used to play from the
# top: libvgm's start() means "back to zero", so starting the transport after the seek undid it
# and this read 0.01 instead of 1.0 (H-1). The tolerance is one block, see the harness help.
host 3 playing
tick
assert engine position ~= 1.0
assert btn.play checked == true

# Scrubbing back before the cue: the song waits at zero again.
host 1 stopped
tick
assert engine position ~= 0
assert btn.play checked == false

# Parked 3 s in with the host stopped: the position follows, the transport does not run.
host 5 stopped
tick
assert engine position ~= 3.0
assert btn.play checked == false
